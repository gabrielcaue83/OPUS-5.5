Para desenvolvedores

Integre em minutos

SDKs nativos, endpoints previsíveis e streaming em tempo real. Uma chamada de API conecta sua aplicação ao núcleo.

  • POST/v1/neural/thinkRaciocínio, análise e geração
  • POST/v1/vision/analyzeVisão computacional em imagens e vídeo
  • GET/v1/streams/liveAnálise de dados em tempo real via SSE
  • POST/v1/agents/orchestrateOrquestração de agentes e automações
from opus import Opus

client = Opus(api_key="SUA_CHAVE")

resposta = client.neural.think(
    model="opus-5.5",
    prompt="Resuma os riscos do contrato",
    modalities=["text", "vision"],
    reasoning="deep",
)
print(resposta.output)  # ~0,4 ms
import Opus from "@opus/sdk";

const client = new Opus({ apiKey: process.env.OPUS_KEY });

const stream = await client.neural.think({
  model: "opus-5.5",
  prompt: "Gere um dashboard de vendas",
  stream: true,
});
for await (const chunk of stream) process.stdout.write(chunk.text);
